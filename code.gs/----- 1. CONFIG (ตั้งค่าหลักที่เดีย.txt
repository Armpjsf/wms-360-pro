/**
 * ฟังก์ชันหลักสำหรับส่งข้อมูลไปยัง Dashboard (Streamlit)
 */
function doGet(e) {
  try {
    const productsData = getProductsData();
    const transactionsData = getTransactionsData();

    const data = {
      products: productsData,
      transactions: transactionsData
    };

    return ContentService
      .createTextOutput(e.parameter.callback + '(' + JSON.stringify(data) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
      
  } catch (err) {
    const errorData = { error: err.message };
    return ContentService
      .createTextOutput(e.parameter.callback + '(' + JSON.stringify(errorData) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
}

/**
 * (ไฟล์ A - Dashboard) ดึงข้อมูลสินค้า [⛔️ แก้ไข ⛔️]
 */
function getProductsData() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(STOCK_SHEET_NAME);
  if (!sheet) throw new Error('ไม่พบชีทชื่อ: "' + STOCK_SHEET_NAME + '"');

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return []; 

  // อ่านข้อมูล A:J (เหมือนเดิม)
  const values = sheet.getRange("A2:J" + lastRow).getValues();
  
  return values
    .filter(function(row) { return row[0]; }) // กรองแถวว่าง (เหมือนเดิม)
    .map(function(row) {
      
      // ----- ⛔️ แก้ไขการแมปคอลัมน์ตรงนี้ ⛔️ -----
      return {
        id: row[0],                      // คอลัมน์ A (ID)
        name: row[0],                    // คอลัมน์ A (ใช้ ID เป็นชื่อด้วย)
        stock: parseInt(row[1], 10) || 0 // ⛔️ คอลัมน์ B (Stock)
      };
      // ----- -------------------------- -----
      
    });
}

/**
 * (ไฟล์ A - Dashboard) ดึงข้อมูล Transaction
 */
function getTransactionsData() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var allTransactions = [];

  // --- ดึงข้อมูลจากชีท "จ่าย" ---
  var issueSheet = ss.getSheetByName(TRANSACTION_SHEET_NAME);
  if (!issueSheet) throw new Error('ไม่พบชีทชื่อ: "' + TRANSACTION_SHEET_NAME + '"');
  
  var issueLastRow = issueSheet.getLastRow();
  if (issueLastRow >= 2) {
    var issueValues = issueSheet.getRange("A2:C" + issueLastRow).getValues();
    var issuedData = issueValues
      .filter(function(row){ return row[0] && row[1]; })
      .map(function(row){
        return {
          date: new Date(row[0]).toISOString().split('T')[0],
          productId: row[1],
          quantity: parseInt(row[2], 10) || 0,
          type: 'จ่าย'
        };
      });
    allTransactions = allTransactions.concat(issuedData);
  }

  // --- ดึงข้อมูลจากชีท "รับ" ---
  var receiveSheet = ss.getSheetByName(RECEIVE_TRANSACTION_SHEET_NAME);
  if (!receiveSheet) throw new Error('ไม่พบชีทชื่อ: "' + RECEIVE_TRANSACTION_SHEET_NAME + '"');

  var receiveLastRow = receiveSheet.getLastRow();
  if (receiveLastRow >= 2) {
    var receiveValues = receiveSheet.getRange("A2:C" + receiveLastRow).getValues();
    var receivedData = receiveValues
      .filter(function(row){ return row[0] && row[1]; })
      .map(function(row){
        return {
          date: new Date(row[0]).toISOString().split('T')[0],
          productId: row[1],
          quantity: parseInt(row[2], 10) || 0,
          type: 'รับ'
        };
      });
    allTransactions = allTransactions.concat(receivedData);
  }

  return allTransactions;
}

function doPost(e) {

  if (event.type === 'message') {
    var userMessage = event.message && event.message.text ? event.message.text : "";

    // ----- 🟢 (ใหม่) บันทึก Log การแชททันที 🟢 -----

    logUserChat(sheet, userId, userMessage,userName ); // 👈 แก้ไขบรรทัดนี้
    // ----- ------------------------------------ -----
    var singleUserMessage = userMessage.trim();
    var lowerCmd = singleUserMessage.toLowerCase();
    

    // ----- ⛔️ เพิ่มส่วนนี้ ⛔️ -----
    // 3.0 ตรวจสอบสถานะ (หมด/เติม)
    // (เราใช้ singleUserMessage เพื่อส่งคำว่า "หมด" หรือ "เติม" ไปค้นหา)

    if (userMessage.indexOf('\n') !== -1) {

            errorLog.push("• " + productName + " -> จำนวนไม่ถูกต้อง");
            continue;

          var matches = searchProduct(stockSheet, productName);

          } else if (matches.length > 1) {
            errorLog.push("• " + productName + " -> พบหลายรายการ โปรดระบุให้ชัดเจนขึ้น");
          } else {
            var product = matches[0];
            var currentStock = product.stock;
            var productRow = product.id; // หมายเลขแถว
            var STOCK_COLUMN_INDEX = 2; // คอลัมน์ B

            // ----- ⛔️ ดึงราคา (เพิ่มใหม่) ⛔️ -----
            var transactionType = isSelling ? 'sell' : 'receive';
            var price = getProductPrice(product.name, transactionType);

            var newRow = [
              formatDate(new Date()), // A: วันที่
              product.name,           // B: รหัสสินค้า
              quantity,               // C: จำนวน
              price,                  // D: ราคา (ใหม่)

                        // ใส่ Location ให้ถูกคอลัมน์
            if (isSelling) {

            if (isSelling) {

              }
            } else {
              var newStock = currentStock + quantity;
              stockSheet.getRange(productRow, STOCK_COLUMN_INDEX).setValue(newStock);
              transactionSheet.appendRow(newRow); 
              successLog.push("✅ " + product.name + " รับเข้า " + quantity + " (คงเหลือ " + newStock + ")");
            }
          }
        } // end for

        var summaryReply = "สรุปผลการ" + (isSelling ? "จ่าย" : "รับ") + "สินค้า:\n";
        if (successLog.length > 0) { summaryReply += "\n--- รายการสำเร็จ ---\n" + successLog.join('\n'); }
        if (errorLog.length > 0) { summaryReply += "\n\n--- เกิดข้อผิดพลาด ---\n" + errorLog.join('\n'); }
        replyText(replyToken, summaryReply);
        return;
      } else {
        // ค้นหาสต็อกแบบหลายบรรทัด
        var summaryReply = "สรุปผลการเช็คสต็อก:\n";
        for (var j = 0; j < lines.length; j++) {

          var matches = searchProduct(stockSheet, keyword);

          } else {
            matches.forEach(function(item) {

            });
          }
        }
        replyText(replyToken, summaryReply);
        return;
      }
    } // end multiline

    // 3.3 คำสั่ง S / R บรรทัดเดียว
    if (singleUserMessage.toUpperCase().startsWith("S ")) {
      var parts = singleUserMessage.split(" ");

      } else if (matches.length === 1) {

      } else {

    } else if (singleUserMessage.toUpperCase().startsWith("R ")) {
      var parts = singleUserMessage.split(" ");
      if (parts.length < 3) {

        return;

    // 3.4 ค้นหาสต็อกแบบบรรทัดเดียว

  // --- ถ้าเป็น postback ให้ไปจัดการ ---
  if (event.type === 'postback') {
    handlePostback(event, sheet, replyToken);
  }
} // end doPost

// HELPER: searchProduct (A=Name, B=Stock, D=Image, Q=Location)
//================================================
function searchProduct(sheet, productName) {
  var matches = [];
  var searchKeyword = productName.trim(); 
  if (searchKeyword === "") return [];

  try {
    var finder = sheet.getRange("A:A").createTextFinder(searchKeyword)
                       .matchEntireCell(false) 
                       .matchCase(false)      
                       .findAll();

    if (finder.length > 0) {
      var maxResults = 10; 
      for (var i = 0; i < Math.min(finder.length, maxResults); i++) {
        var cell = finder[i];
        var row = cell.getRow();
        var productData = sheet.getRange(row, 1, 1, 17).getValues()[0]; 
        
        matches.push({
          name: productData[0], // A = Name
          stock: productData[1], // B = Stock
          imageUrl: cleanImageUrl(productData[3]), // D = Image (ดึงผ่านตัวดึงภาพ Thumbnail อัตโนมัติ)
          location: productData[16] || "N/A", // Q = Location (index 16)
          id: row // 'id' คือหมายเลขแถว
        });
      }
    }
    return matches;
  } catch (e) {
    return []; 
  }
}

// HELPER: handlePostback
//================================================
function handlePostback(event, sheet, replyToken) {
  var data = event.postback && event.postback.data ? event.postback.data : "";
  var stockSheet = sheet.getSheetByName(STOCK_SHEET_NAME);
  
  var STOCK_COLUMN_INDEX = 2; // (คอลัมน์ B)
  var LOCATION_COLUMN_INDEX = 17; // (คอลัมน์ Q => index 17)

  function createNewRow(date, name, unit, qty, location, transactionType, price) { // 👈 7. เพิ่ม price
  var row = [
    formatDate(date), // A: วันที่
    name,             // B: รหัสสินค้า
    qty,              // C: จำนวน
    price,            // D: ราคา (ใหม่)
    unit,             // E: หน่วยนับ
      "",               // F
      "",               // G
      "",               // H
      "",               // I (ที่ตั้ง 'รับ')
      "",               // J
      "",               // K
      "",               // L
      ""                // M (ที่ตั้ง 'จ่าย')
    ];
    
    // ใส่ Location ให้ถูกคอลัมน์
    if (transactionType === 'sell') {
      row[12] = location; // Index 12 คือ Column M
    } else { // 'receive'
      row[8] = location;  // Index 8 คือ Column I
    }
    
    return row;
  }

  if (data.indexOf("confirm_sell:") === 0) {
    var parts = data.split(':');
    var productId = parseInt(parts[1], 10);
    var quantity = parseInt(parts[2], 10);
    var transactionSheet = sheet.getSheetByName(TRANSACTION_SHEET_NAME);

    try {
      var productData = stockSheet.getRange(productId, 1, 1, LOCATION_COLUMN_INDEX).getValues()[0];
      var productName = productData[0];
      var currentStock = productData[STOCK_COLUMN_INDEX - 1];
      var productLocation = productData[LOCATION_COLUMN_INDEX - 1];
      
      if (currentStock < quantity) {
        replyText(replyToken, "❌ ไม่สามารถขายได้ เนื่องจากสินค้ามีเพียง " + currentStock + " ชิ้น");
        return;
      }
      
      var price = getProductPrice(productName, 'sell'); // 👈 เพิ่มบรรทัดนี้
var newRow = createNewRow(new Date(), productName, "แผ่น", quantity, productLocation, 'sell', price); // 👈 แก้บรรทัดนี้
      transactionSheet.appendRow(newRow);
      
      var newStock = currentStock - quantity;
      stockSheet.getRange(productId, STOCK_COLUMN_INDEX).setValue(newStock);
      replyText(replyToken, "✅ บันทึกการขาย " + productName + " จำนวน " + quantity + " ชิ้นเรียบร้อยแล้ว");
    } catch (e) {
      replyText(replyToken, "⚠️ เกิดข้อผิดพลาด: " + e.message);
    }
  } else if (data.indexOf("confirm_receive:") === 0) {
    var parts = data.split(':');
    var productId = parseInt(parts[1], 10);
    var quantity = parseInt(parts[2], 10);
    var receiveTransactionSheet = sheet.getSheetByName(RECEIVE_TRANSACTION_SHEET_NAME);

    try {
      var productData = stockSheet.getRange(productId, 1, 1, LOCATION_COLUMN_INDEX).getValues()[0];
      var productName = productData[0];
      var currentStock = productData[STOCK_COLUMN_INDEX - 1];
      var productLocation = productData[LOCATION_COLUMN_INDEX - 1];
      
      var price = getProductPrice(productName, 'receive'); // 👈 เพิ่มบรรทัดนี้
var newRow = createNewRow(new Date(), productName, "แผ่น", quantity, productLocation, 'receive', price); // 👈 แก้บรรทัดนี้
      receiveTransactionSheet.appendRow(newRow);
      
      var newStock = currentStock + quantity;
      stockSheet.getRange(productId, STOCK_COLUMN_INDEX).setValue(newStock);
      replyText(replyToken, "✅ บันทึกการรับ " + productName + " จำนวน " + quantity + " ชิ้นเรียบร้อยแล้ว");
    } catch (e) {
      replyText(replyToken, "⚠️ เกิดข้อผิดพลาด: " + e.message);
    }
  }
}

//================================================
function getProductPrice(productName, transactionType) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var productSheet = ss.getSheetByName(PRODUCT_MASTER_SHEET_NAME);
    if (!productSheet) {
      Logger.log("ไม่พบชีท 'ชื่อสินค้า'");
      return ""; // ไม่พบชีท
    }

    // ค้นหา productName ในคอลัมน์ A (ชื่อสินค้า)
    var finder = productSheet.getRange("A:A").createTextFinder(productName)
                             .matchEntireCell(true) // ค้นหาแบบตรงตัว
                             .findNext();
    
    if (finder) {
      var row = finder.getRow();
      var priceColIndex;
      
      if (transactionType === 'receive') {
        priceColIndex = 3; // 💸 รับ (ดึงจากคอลัมน์ C)
      } else { // 'sell'
        priceColIndex = 4; // 💰 จ่าย (ดึงจากคอลัมน์ D)
      }
      
      var price = productSheet.getRange(row, priceColIndex).getValue();
      return price;
      
    } else {
      Logger.log("ไม่พบราคาสำหรับสินค้า: " + productName + " ในชีท 'ชื่อสินค้า'");
      return ""; // ไม่พบสินค้า
    }
  } catch (e) {
    Logger.log("Error in getProductPrice: " + e.message);
    return ""; // เกิดข้อผิดพลาด
  }
}

function formatDate(date) {
  var d = new Date(date);
  var day = ('0' + d.getDate()).slice(-2);
  var monthNames = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  var month = monthNames[d.getMonth()];
  var year = d.getFullYear();
  return day + "/" + month + "/" + year;
}

function searchAndReportMovement(typeToSearch, replyToken) {

      foundItems.push(itemName + " : " + stockQty + " ชิ้น");
    }
  }
  sendMovementReport(typeToSearch, foundItems, replyToken);
}

// ⛔️ ฟังก์ชันใหม่: ค้นหาสถานะ (คอลัมน์ E) [อัปเดตล่าสุด]
//================================================
function searchAndReportStatus(statusToSearch, replyToken) {
  var sheet = SpreadsheetApp.openById(SPREADSHEET_ID);
  var stockSheet = sheet.getSheetByName(STOCK_SHEET_NAME);
  if (!stockSheet) {
    replyText(replyToken, "ไม่พบชีทสินค้า: " + STOCK_SHEET_NAME);
    return;
  }
  
  // ----- ⛔️ 1. ส่วนที่เพิ่ม: การจับคู่คำ ⛔️ -----
  var command = statusToSearch.toLowerCase().trim(); // สิ่งที่ผู้ใช้พิมพ์
  var actualSearchTerm = ""; // สิ่งที่บอทจะใช้ค้นหาในชีท

  if (command === "หมด") {
    actualSearchTerm = "สินค้าหมด";
  } else if (command === "เติม") {
    actualSearchTerm = "ควรเติมทันที";
  } else {
    // กรณีนี้ไม่ควรเกิดขึ้น ถ้า doPost ทำงานถูกต้อง
    actualSearchTerm = command; 
  }
  
  // แปลงคำค้นหาจริงเป็นตัวพิมพ์เล็ก (สำหรับการเปรียบเทียบแบบไม่สน case)
  var lowerActualSearchTerm = actualSearchTerm.toLowerCase();
  // ----- --------------------------------- -----

  
  var allData = stockSheet.getRange("A2:J" + stockSheet.getLastRow()).getValues(); 
  var foundItems = [];
  
  const NAME_IDX = 0; // คอลัมน์ A (ชื่อสินค้า)
  const STOCK_QTY_IDX = 1; // คอลัมน์ B (จำนวนคงเหลือ)
  const STATUS_IDX = 4; // คอลัมน์ E (index 4)
  
  for (var i = 0; i < allData.length; i++) { 
    var row = allData[i];
    var statusCol = String(row[STATUS_IDX] || '').trim(); // ค่าจากคอลัมน์ E
    
    // ⛔️ 2. ส่วนที่แก้ไข: ใช้คำค้นหาที่จับคู่ไว้ (lowerActualSearchTerm)
    // (ใช้ .indexOf() !== -1 เพื่อค้นหาว่า "มีคำนี้อยู่หรือไม่")
    if (statusCol.toLowerCase().indexOf(lowerActualSearchTerm) !== -1) {
      
      var itemName = String(row[NAME_IDX] || '').trim();
      var stockQty = row[STOCK_QTY_IDX]; // B
      foundItems.push("• " + itemName + " : " + stockQty + " ชิ้น");
    }
  }
  
  // ⛔️ 3. ส่วนที่ยืนยัน: ส่งรายงานโดยใช้ "คำสั่งเดิม" ของผู้ใช้
  // (เพื่อให้หัวข้อรายงานแสดงว่า "หมด" หรือ "เติม" ไม่ใช่ "สินค้าหมด")
  sendStatusReport(statusToSearch, foundItems, replyToken);
}

/**
 * ฟังก์ชันตรวจสอบและเติมสกุลไฟล์รูปภาพ (.jpg) ต่อท้ายลิงก์ของ Google 
 * เพื่อแก้ปัญหา LINE เวอร์ชั่นใหม่อ่านรูปภาพไม่ขึ้น
 */
function cleanImageUrl(url) {
  if (!url || typeof url !== 'string') {
    return "https://via.placeholder.com/512x512.png?text=No+Image";
  }
  
  var cleanUrl = url.trim();
  if (cleanUrl === "") {
    return "https://via.placeholder.com/512x512.png?text=No+Image";
  }
  
  // 1. ถ้าลิงก์ในคอลัมน์ D เป็น https://lh3.googleusercontent.com/d/... อยู่แล้ว
  if (cleanUrl.indexOf("lh3.googleusercontent.com/d/") !== -1) {
    // ตรวจสอบว่ามีการใส่ ?.jpg ไปหรือยัง ถ้ายังไม่มีให้เติมเข้าไปท้ายลิงก์ทันที
    if (cleanUrl.indexOf("?.jpg") === -1 && cleanUrl.indexOf("?.png") === -1) {
      return cleanUrl + "?.jpg";
    }
    return cleanUrl;
  }
  
  // 2. กรณีถ้าหลุดมาเป็นลิงก์ Google Drive แบบแชร์ธรรมดา ให้แปลงเป็น lh3 และเติม ?.jpg ให้ด้วย
  var driveIdMatch = cleanUrl.match(/\/file\/d\/([a-zA-Z0-9_-]+)/) || cleanUrl.match(/[?&]id=([a-zA-Z0-9_-]+)/);
  if (driveIdMatch && driveIdMatch[1]) {
    var fileId = driveIdMatch[1];
    return "https://lh3.googleusercontent.com/d/" + fileId + "?.jpg";
  }
  
  return cleanUrl;
}
